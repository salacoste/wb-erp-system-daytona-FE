**Type definitions to create:**

```typescript
// Sort/group enums
export type SearchOrderBy = 'totalImpressions' | 'totalClicks' | 'avgPosition' | 'avgCtr' | 'totalOrders' | 'totalRevenue'
export type SearchOrdersGroupBy = 'query' | 'product' | 'day'

// Shared period type
export interface SearchPeriod { from: string; to: string }

// --- By-Product ---
export interface SearchByProductParams {
  nmId: number       // REQUIRED
  from: string       // REQUIRED (YYYY-MM-DD)
  to: string         // REQUIRED (YYYY-MM-DD)
  orderBy?: SearchOrderBy  // default: totalImpressions
  limit?: number     // default: 50, max: 100
}

export interface SearchQueryItem {
  searchQuery: string
  avgPosition: number
  totalImpressions: number
  totalClicks: number
  avgCtr: number
  totalOrders: number
  totalRevenue: number  // May always be 0 — backend limitation
}

export interface SearchByProductResponse {
  nmId: number
  period: SearchPeriod
  queries: SearchQueryItem[]
  totalQueries: number
}

// --- By-Query ---
export interface SearchByQueryParams {
  query: string      // REQUIRED, ILIKE partial match
  from: string
  to: string
  limit?: number     // default: 50, max: 100
}

export interface SearchProductItem {
  nmId: number
  vendorCode: string | null
  avgPosition: number
  totalImpressions: number
  totalClicks: number
  avgCtr: number
  totalOrders: number
  totalRevenue: number  // May always be 0
}

export interface SearchByQueryResponse {
  query: string
  period: SearchPeriod
  products: SearchProductItem[]
  totalProducts: number
}

// --- Orders ---
export interface SearchOrdersParams {
  from: string
  to: string
  groupBy?: SearchOrdersGroupBy  // default: 'query'
  limit?: number                 // default: 50, max: 100
}

export interface SearchOrderItem {
  key: string         // query text, nmId as string, or date string — depends on groupBy
  totalOrders: number
  totalRevenue: number
  uniqueProducts?: number   // present when groupBy='query'
  uniqueQueries?: number    // present when groupBy='product'
}

export interface SearchOrdersSummary {
  totalSearchOrders: number
  totalSearchRevenue: number
  searchOrderShare: number    // percentage 0-100
}

export interface SearchOrdersResponse {
  period: SearchPeriod
  groupBy: SearchOrdersGroupBy
  items: SearchOrderItem[]
  summary: SearchOrdersSummary
}
```
