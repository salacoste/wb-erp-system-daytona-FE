esac

case "$HANDOFF_PHASE" in
  delivery | publish-recovery)
    test ! -L "$HANDOFF_WORKTREE"
    test -e "$HANDOFF_WORKTREE"
    if ! ACTIVE_HANDOFF_TOPLEVEL="$(git -C "$HANDOFF_WORKTREE" \
      rev-parse --show-toplevel)"; then
      exit 71
    fi
    if ! ACTIVE_HANDOFF_BRANCH="$(git -C "$HANDOFF_WORKTREE" \
      branch --show-current)"; then
      exit 71
    fi
    test "$ACTIVE_HANDOFF_TOPLEVEL" = "$HANDOFF_WORKTREE"
    test "$ACTIVE_HANDOFF_BRANCH" = "$HANDOFF_BRANCH"
    ;;
  cleanup)
    test ! -L "$HANDOFF_WORKTREE"
    if test -e "$HANDOFF_WORKTREE"; then
      if ! CLEANUP_HANDOFF_TOPLEVEL="$(git -C "$HANDOFF_WORKTREE" \
        rev-parse --show-toplevel)"; then

      fi
      if ! CLEANUP_HANDOFF_BRANCH="$(git -C "$HANDOFF_WORKTREE" \

        exit 71
      fi
      test "$CLEANUP_HANDOFF_TOPLEVEL" = "$HANDOFF_WORKTREE"
      test "$CLEANUP_HANDOFF_BRANCH" = "$HANDOFF_BRANCH"
    else
      git -C "$FRONTEND_REPOSITORY" worktree prune
      if ! HANDOFF_WORKTREE_INVENTORY="$(git -C "$FRONTEND_REPOSITORY" \
        worktree list --porcelain)"; then
        exit 71
      fi
      case "$HANDOFF_WORKTREE_INVENTORY" in
        *"$HANDOFF_WORKTREE"*) exit 65 ;;
      esac
    fi
    ;;
esac

extract_handoff_payload() {
  HANDOFF_SOURCE_FILE="$1"
  HANDOFF_MARKER_START="$2"
  HANDOFF_MARKER_END="$3"
  HANDOFF_OUTPUT_FILE="$4"
  if ! awk -v marker_start="$HANDOFF_MARKER_START" \
    -v marker_end="$HANDOFF_MARKER_END" '
      $0 == marker_start {
        if (state != 0 || start_count != 0) exit 2
        start_count = 1
        state = 1
        next
      }
      $0 == marker_end {
        if (state != 1 || end_count != 0) exit 2
        end_count = 1
        state = 2
        next
      }
      state == 1 { print }
      END {
        if (start_count != 1 || end_count != 1 || state != 2) exit 2
      }
    ' "$HANDOFF_SOURCE_FILE" > "$HANDOFF_OUTPUT_FILE"; then
    return 71
  fi
  test -s "$HANDOFF_OUTPUT_FILE"
}

assert_handoff_wip_semantics() {
  ASSERTED_HANDOFF_MAIN_MODE="$1"
  ASSERTED_HANDOFF_WIP_FILE="$2"
  ASSERTED_HANDOFF_WIP_SHA256="$3"
  AUTHENTICATED_CLEANUP_WIP_SHA256="$4"
  test -s "$ASSERTED_HANDOFF_WIP_FILE" || return 64
  COMPUTED_ASSERTED_HANDOFF_WIP_SHA256="$(shasum -a 256 \
    "$ASSERTED_HANDOFF_WIP_FILE" | awk '{ print $1 }')" || return 71
  test "$COMPUTED_ASSERTED_HANDOFF_WIP_SHA256" = \
    "$ASSERTED_HANDOFF_WIP_SHA256" || return 64
  test "$ASSERTED_HANDOFF_WIP_SHA256" = \
    "$AUTHENTICATED_CLEANUP_WIP_SHA256" || return 64
  case "$ASSERTED_HANDOFF_MAIN_MODE" in
    clean-local-main)
      cmp -s "$ASSERTED_HANDOFF_WIP_FILE" <(printf 'NONE\n') || return 64
      ;;
    recorded-origin-deferral)
      if awk 'NF == 0 { blank = 1 } END { exit blank ? 0 : 1 }' \
        "$ASSERTED_HANDOFF_WIP_FILE"; then
        return 64
      fi
      LC_ALL=C sort -u "$ASSERTED_HANDOFF_WIP_FILE" > \
        "$HANDOFF_TMP/canonical-asserted-handoff-wip" || return 71
      cmp -s "$ASSERTED_HANDOFF_WIP_FILE" \
        "$HANDOFF_TMP/canonical-asserted-handoff-wip" || return 64
      ;;
    *) return 64 ;;
  esac
}

## Mandatory cleanup

Run the cleanup fence first with `STORY_169_14_CLEANUP_PHASE=branch-worktree`. After authenticating the exact backend-main mode and byte-exact cleanup reservation, that phase atomically publishes the strict nine-line mode-600 `story-169-14-branch-worktree-cleanup-authority-v1` before deleting the backend remote branch, worktree, or local branch. Publication is create-or-byte-identical, so a crash before or after any deletion restarts from the same reviewed-record hashes, branch/worktree identity, mode, and reservation hash. After the exact artifact-only final-handoff commit has merged into frontend `main`, rerun the same fence with `STORY_169_14_CLEANUP_PHASE=record-retirement`; create-worktree preflight never consumes this cleanup-only input. Immediately before deleting any of the five source records, retirement atomically publishes the strict 48-line mode-600 `story-169-14-record-retirement-transaction-v1`, binding their exact paths and SHA-256 hashes, both merged PRs, refreshed-main ancestry, completed branch/worktree cleanup, and the exact deletion set. A rerun first reauthenticates the exact single fetch/push remote identity for both repositories, both PR base/head topologies, local/remote branch absence, and worktree-inventory absence. Before considering whether a source exists, it validates every transaction-bound source SHA-256 as exactly 64 lowercase hexadecimal characters. Every source is then accepted only as present with that validated transaction-bound hash or already absent under the authenticated transaction; a malformed bound hash fails closed in either state, and executable synthetics cover malformed and valid hashes with the source both present and absent. The fence deletes the remainder idempotently, proves all five absent, then deletes and proves absence of the transaction record. Story 169.15 preflight independently reconstructs and hashes the retired cleanup authority, rejects all five source records and any transaction residue, and treats a dangling symlink at every expected-absent record/bootstrap/transaction/worktree path as residue rather than absence.
